async function generateUrl() { const input = document.getElementById('fileInput'); const status = document.getElementById('status'); const urlResult = document.getElementById('urlResult'); if (!input.files[0]) { status.innerHTML = 'Fayl tanlanmadi!'; return; } status.innerHTML = 'Yuklanmoqda...'; const formData = new FormData(); formData.append('file', input.files[0]); try { const response = await fetch('https://tmpfiles.org/api/v1/upload', { method: 'POST', body: formData }); const data = await response.json(); if (data.status === 'success' && data.data && data.data.url) { // To'g'ridan-to'g'ri yuklanadigan URL manziliga o'tkazamiz const directUrl = data.data.url.replace('tmpfiles.org/', 'tmpfiles.org/dl/'); urlResult.value = directUrl; status.innerHTML = 'Tayyor!'; saveToHistory(directUrl); } else { status.innerHTML = 'Serverda xatolik yuz berdi!'; } } catch (e) { status.innerHTML = 'Ulanishda xatolik!'; } }
